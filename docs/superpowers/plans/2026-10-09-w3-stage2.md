```ts
/**
 * Responsive classes for the Floating Canvas (W3, spec §2.5). Width 0 is what
 * the static render sees, so it must be a phone: the HTML is phone-first and
 * wider screens re-lay out after hydration.
 */
export type LayoutClass = 'phone' | 'tablet' | 'desktop';

export const TABLET_MIN = 600;
export const DESKTOP_MIN = 1024;

export function layoutClassFor(width: number): LayoutClass {
  if (!(width >= TABLET_MIN)) return 'phone';
  return width >= DESKTOP_MIN ? 'desktop' : 'tablet';
}

/** Grid columns per layout. Desktop uses the tablet grid until Stage 3. */
export function columnsFor(layout: LayoutClass, _kind: 'forYou' | 'covers'): number {
  return layout === 'phone' ? 2 : 4;
}
```

```ts
export function useLayoutClass(): LayoutClass {
```

```ts
/** Static stage for entity pages: cover colour at the top, canvas at the bottom; neutral without a palette. */
export function entityStageGradient(palette: ArtworkPalette | undefined): { colors: string[]; locations: number[] } {
  const top = palette ? stageTop(palette) : Colors.surface.l1;
  return { colors: [top, mixHex(top, Colors.canvas, 0.55), Colors.canvas], locations: [0, 0.55, 1] };
}
```

```ts
export type Rect = { x: number; y: number; width: number; height: number };

/**
 * Popover position for the capsule ⋯ menu (W3 spec §2.3): above the anchor —
 * and above anything in `avoid` under it — when it fits, else below; always
 * inside the window, height-limited when the window is short.
 */
export function placeMenu(anchor: Rect, menu: { width: number; height: number }, viewport: { width: number; height: number }, avoid: Rect[], margin = 8): { x: number; y: number; maxHeight: number } {
  const x = Math.min(Math.max(anchor.x + anchor.width - menu.width, margin), viewport.width - menu.width - margin);
  const under = avoid.filter((r) => r.x < x + menu.width && x < r.x + r.width).map((r) => r.y);
  const floor = Math.min(anchor.y, ...under) - margin;
  const above = floor - margin;
  const below = viewport.height - (anchor.y + anchor.height) - 2 * margin;
  if (menu.height <= above || above >= below) {
    const maxHeight = Math.min(menu.height, above);
    return { x, y: floor - maxHeight, maxHeight };
  }
  return { x, y: anchor.y + anchor.height + margin, maxHeight: Math.min(menu.height, below) };
}
```
